** How =callPackage= Solves This

In nixpkgs, the package set is built roughly like this:
#+begin_src nix

    foo = self.callPackage ./foo.nix {};

    baz = self.callPackage ./baz.nix {};  # can depend on foo and bar
  });
in pkgs
#+end_src

1. *Lazy evaluation*: Dependencies aren't evaluated until actually needed
2. *Fixed-point*: =lib.makeScope= creates a fixed-point where =self= eventually equals the final result
3. *Automatic injection*: Each package function gets the full package set injected

let
  topoSort = plugins:
    let
      # Create a lookup map from plugin name to plugin
      pluginMap = builtins.listToAttrs (
        map (plugin: { 
          name = plugin.name; 
          value = plugin; 
        }) plugins
      );
      
      # DFS-based topological sort with cycle detection
      visit = visited: visiting: plugin:
        let
          pluginName = plugin.name;
          deps = plugin.rapturePluginInputs or [];
        in
          if builtins.elem pluginName visiting then
            throw "Circular dependency detected involving plugin: ${pluginName}"
          else if builtins.elem pluginName visited then
            # Already processed, return empty result
            { inherit visited; result = []; }
          else
            let
              newVisiting = visiting ++ [pluginName];
              
              # Visit all dependencies first
              visitDependency = acc: depName:
                if pluginMap ? ${depName} then
                  let
                    depResult = visit acc.visited newVisiting pluginMap.${depName};
                  in {
                    visited = depResult.visited;
                    result = acc.result ++ depResult.result;
                  }
                else
                  throw "Plugin dependency '${depName}' not found. Required by '${pluginName}'";
              
              depResults = builtins.foldl' visitDependency 
                { visited = visited; result = []; } 
                deps;
            in {
              visited = depResults.visited ++ [pluginName];
              result = depResults.result ++ [plugin];
            };
      
      # Visit all plugins, avoiding duplicates in final result
      processAllPlugins = plugins:
        let
          processPlugin = acc: plugin:
            let
              result = visit acc.visited [] plugin;
              # Filter out plugins already in the result to avoid duplicates
              newPlugins = builtins.filter 
                (p: !(builtins.any (existing: existing.name == p.name) acc.result))
                result.result;
            in {
              visited = result.visited;
              result = acc.result ++ newPlugins;
            };

          (builtins.foldl' processPlugin { visited = []; result = []; } plugins).result;

      processAllPlugins plugins;
