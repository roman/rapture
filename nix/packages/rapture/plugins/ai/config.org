#+begin_src emacs-lisp :tangle yes
  (use-package mcp
    :after (evil gptel)
    :init
    (defun rapture/mcp-extract-servers-from-json (json-file)
      "Extract MCP server configuration from JSON-FILE, set `mcp-hub-servers' variable and display a message."
      (interactive "fSelect JSON file: ")
      (with-temp-buffer
        (insert-file-contents json-file)
        (let* ((json-object-type 'hash-table)
  	     (json-array-type 'list)  ;; Ensure JSON arrays become lists
  	     (json-key-type 'string)
  	     (json-data (json-read-from-string (buffer-string)))
  	     (servers (gethash "mcpServers" json-data))
  	     (server-configs '()))

  	(maphash (lambda (name config)
  		   (let* ((command (gethash "command" config))
  			  (args (gethash "args" config))
  			  (env (gethash "env" config))
  			  (server-plist nil))

  		     ;; Add command
  		     (setq server-plist (plist-put server-plist :command command))

  		     ;; Add args if present and non-empty (as a list, not vector)
  		     (when (and args (> (length args) 0))
  		       (setq server-plist (plist-put server-plist :args args)))

  		     ;; Add env if present and non-empty
  		     (let ((env-plist '()))
  		       (when env
  			 (maphash (lambda (k v)
  				    (setq env-plist (plist-put env-plist
  							       (intern (concat ":" k))
  							       v)))
  				  env)
  			 (when env-plist
  			   (setq server-plist (plist-put server-plist :env env-plist)))))

  		     ;; Add server to list
  		     (push (cons name server-plist) server-configs)))
  		 servers)

  	;; Generate the setq form and evaluate it
  	(let* ((server-config-list (reverse server-configs))
  	       (server-count (length server-config-list)))

  	  ;; Evaluate the setq statement
  	  (setq mcp-hub-servers server-config-list)

  	  ;; Just message the result instead of showing a buffer
  	  (message "Extracted and set %d MCP Hub Servers from %s"
  		   server-count
  		   (file-name-nondirectory json-file))))))

    (defun rapture/gptel-mcp-register-tool ()
      (interactive)
      (let ((tools (mcp-hub-get-all-tool :asyncp t :categoryp t)))
        (mapcar #'(lambda (tool)
    		  (apply #'gptel-make-tool
    			 tool))
    	      tools)))

    (defun rapture/gptel-mcp-use-tool ()
      (interactive)
      (let ((tools (mcp-hub-get-all-tool :asyncp t :categoryp t)))
        (mapcar #'(lambda (tool)
    		  (let ((path (list (plist-get tool :category)
    				    (plist-get tool :name))))
    		    (push (gptel-get-tool path)
    			  gptel-tools)))
    	      tools)))

    (defun rapture/gptel-mcp-close-use-tool ()
      (interactive)
      (let ((tools (mcp-hub-get-all-tool :asyncp t :categoryp t)))
        (mapcar #'(lambda (tool)
    		  (let ((path (list (plist-get tool :category)
    				    (plist-get tool :name))))
    		    (setq gptel-tools
    			  (cl-remove-if #'(lambda (tool)
    					    (equal path
    						   (list (gptel-tool-category tool)
    							 (gptel-tool-name tool))))
    					gptel-tools))))
    	      tools)))

    :config
    (require 'gptel-integrations)
    (require 'mcp-hub)
    (evil-define-key 'normal 'global (kbd "SPC i h f") #'rapture/mcp-extract-servers-from-json)
    (evil-define-key 'normal 'global (kbd "SPC i h h") #'mcp-hub)
    (evil-define-key 'normal 'global (kbd "SPC i h S") #'mcp-hub-start-all-server)
    (evil-define-key 'normal 'global (kbd "SPC i h l") #'mcp-hub-view-log)
    )
#+end_src
