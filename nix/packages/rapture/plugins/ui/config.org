** Fancy Splash Window

#+BEGIN_SRC emacs-lisp :tangle yes
  (setq initial-scratch-message nil)
  (setq inhibit-startup-screen t)

  (defun rapture/show-welcome-buffer ()
    "Show *Welcome* buffer."
    (with-current-buffer (get-buffer-create "*Welcome*")
      (setq truncate-lines t)
      (let* ((buffer-read-only)
	     (image-path "@splash@")
	     (image (create-image image-path))
	     (size (image-size image))
	     (height (cdr size))
	     (width (car size))
	     (top-margin (floor (/ (- (window-height) height) 2)))
	     (left-margin (floor (/ (- (window-width) width) 2)))
	     (prompt-title "Welcome to Emacs!"))
	(erase-buffer)
	(setq mode-line-format nil)
	(goto-char (point-min))
	(insert (make-string top-margin ?\n ))
	(insert (make-string left-margin ?\ ))
	(insert-image image)
	(insert "\n\n\n")
	(insert (make-string (floor (/ (- (window-width) (string-width prompt-title)) 2)) ?\ ))
	(insert prompt-title))
      (setq cursor-type nil)
      (read-only-mode +1)
      (switch-to-buffer (current-buffer))
      ;; (local-set-key (kbd "q") 'kill-this-buffer)
      (evil-local-set-key 'normal (kbd "q") #'kill-this-buffer)
      (evil-local-set-key 'insert (kbd "q") #'kill-this-buffer)))

  (when (< (length command-line-args) 2)
    (add-hook 'emacs-startup-hook (lambda ()
				    (when (display-graphic-p)
				      (rapture/show-welcome-buffer)))))
#+END_SRC

** Ligatures

I want pretty symbols on my code

#+BEGIN_SRC emacs-lisp :tangle yes
  (use-package ligature
    :config
    ;; Enable the "www" ligature in every possible major mode
    (ligature-set-ligatures 't '("www"))
    ;; Enable traditional ligature support in eww-mode, if the
    ;; `variable-pitch' face supports it
    (ligature-set-ligatures 'eww-mode '("ff" "fi" "ffi"))
    ;; Enable all Cascadia and Fira Code ligatures in programming modes
    (ligature-set-ligatures 'prog-mode
			  '(;; == === ==== => =| =>>=>=|=>==>> ==< =/=//=// =~
			    ;; =:= =!=
			    ("=" (rx (+ (or ">" "<" "|" "/" "~" ":" "!" "="))))
			    ;; ;; ;;;
			    (";" (rx (+ ";")))
			    ;; && &&&
			    ("&" (rx (+ "&")))
			    ;; !! !!! !. !: !!. != !== !~
			    ("!" (rx (+ (or "=" "!" "\." ":" "~"))))
			    ;; ?? ??? ?:  ?=  ?.
			    ("?" (rx (or ":" "=" "\." (+ "?"))))
			    ;; %% %%%
			    ("%" (rx (+ "%")))
			    ;; |> ||> |||> ||||> |] |} || ||| |-> ||-||
			    ;; |->>-||-<<-| |- |== ||=||
			    ;; |==>>==<<==<=>==//==/=!==:===>
			    ("|" (rx (+ (or ">" "<" "|" "/" ":" "!" "}" "\]"
					    "-" "=" ))))
			    ;; \\ \\\ \/
			    ("\\" (rx (or "/" (+ "\\"))))
			    ;; ++ +++ ++++ +>
			    ("+" (rx (or ">" (+ "+"))))
			    ;; :: ::: :::: :> :< := :// ::=
			    (":" (rx (or ">" "<" "=" "//" ":=" (+ ":"))))
			    ;; // /// //// /\ /* /> /===:===!=//===>>==>==/
			    ("/" (rx (+ (or ">"  "<" "|" "/" "\\" "\*" ":" "!"
					    "="))))
			    ;; .. ... .... .= .- .? ..= ..<
			    ("\." (rx (or "=" "-" "\?" "\.=" "\.<" (+ "\."))))
			    ;; -- --- ---- -~ -> ->> -| -|->-->>->--<<-|
			    ("-" (rx (+ (or ">" "<" "|" "~" "-"))))
			    ;; *> */ *)  ** *** ****
			    ("*" (rx (or ">" "/" ")" (+ "*"))))
			    ;; www wwww
			    ("w" (rx (+ "w")))
			    ;; <> <!-- <|> <: <~ <~> <~~ <+ <* <$ </  <+> <*>
			    ;; <$> </> <|  <||  <||| <|||| <- <-| <-<<-|-> <->>
			    ;; <<-> <= <=> <<==<<==>=|=>==/==//=!==:=>
			    ;; << <<< <<<<
			    ("<" (rx (+ (or "\+" "\*" "\$" "<" ">" ":" "~"  "!"
					    "-"  "/" "|" "="))))
			    ;; >: >- >>- >--|-> >>-|-> >= >== >>== >=|=:=>>
			    ;; >> >>> >>>>
			    (">" (rx (+ (or ">" "<" "|" "/" ":" "=" "-"))))
			    ;; #: #= #! #( #? #[ #{ #_ #_( ## ### #####
			    ("#" (rx (or ":" "=" "!" "(" "\?" "\[" "{" "_(" "_"
					 (+ "#"))))
			    ;; ~~ ~~~ ~=  ~-  ~@ ~> ~~>
			    ("~" (rx (or ">" "=" "-" "@" "~>" (+ "~"))))
			    ;; __ ___ ____ _|_ __|____|_
			    ("_" (rx (+ (or "_" "|"))))
			    ;; Fira code: 0xFF 0x12
			    ("0" (rx (and "x" (+ (in "A-F" "a-f" "0-9")))))
			    ;; Fira code:
			    "Fl"  "Tl"  "fi"  "fj"  "fl"  "ft"
			    ;; The few not covered by the regexps.
			    "{|"  "[|"  "]#"  "(*"  "}#"  "$>"  "^="))
    ;; Enables ligature checks globally in all buffers. You can also do it
    ;; per mode with `ligature-mode'.
    (global-ligature-mode t))
#+END_SRC

    (use-package highlight-indent-guides
      :init
      ;; https://emacs.stackexchange.com/a/74191
      (defun rapture/-dump-face-attributes (face &rest props)
	(cl-loop for prop in (or props
				 (sort (mapcar #'car face-attribute-name-alist)
				       (lambda (s1 s2)
					 (string< (symbol-name s1) (symbol-name s2)))))
		 for val = (face-attribute face prop nil t)
		 unless (eq val 'unspecified)
		 append (list prop val)))
      (defun rapture/get-face-attribute (face prop)
	(plist-get (rapture/-dump-face-attributes face prop) prop))

      :config
      ;; disable default coloring
      (setq highlight-indent-guides-auto-enabled nil)
      (set-face-background
       'highlight-indent-guides-even-face
       (rapture/get-face-attribute 'font-lock-type-face :foreground))

      (set-face-background
	'highlight-indent-guides-odd-face
	(rapture/get-face-attribute 'font-lock-variable-name-face :foreground))

      ;; ensure we are using the column style
      (setq highlight-indent-guides-method 'column)

	'normal prog-mode-map (kbd "SPC t i")

** Search Utilities (consult)

Use embark to export consult searches and modify multiple files at once

#+BEGIN_SRC emacs-lisp :tangle yes
  (use-package wgrep)

  (use-package embark
    :after (wgrep)
    :init
    (setq prefix-help-command #'embark-prefix-help-command)

    (defun +embark/export-write ()
      "Export the current vertico results to a writable buffer if possible.
       Supports exporting consult-grep to wgrep, file to wdeired, and consult-location to occur-edit"
      (interactive)
      (require 'embark)
      (let* ((edit-command
	      (pcase-let ((`(,type . ,candidates)
			   (run-hook-with-args-until-success 'embark-candidate-collectors)))
		(pcase type
		  ('consult-grep #'wgrep-change-to-wgrep-mode)
		  ('file #'wdired-change-to-wdired-mode)
		  ('consult-location #'occur-edit-mode)
		  (x (user-error "embark category %S doesn't support writable export" x)))))
	     (embark-after-export-hook `(,@embark-after-export-hook ,edit-command)))
	(embark-export)))

    :bind
    (:map minibuffer-mode-map
	  ("C-c C-e" . +embark/export-write)
	  ("C-c C-;" . embark-export)
	  ("C-c C-l" . embark-collect))
    :config
    ;; Hide the mode line of the Embark live/completions buffers
    (add-to-list 'display-buffer-alist
		 '("\\`\\*Embark Collect \\(Live\\|Completions\\)\\*"
		   nil
		   (window-parameters (mode-line-format . none)))))
#+END_SRC
