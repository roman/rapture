* Introduction

After transition from vanilla emacs to [[https://www.spacemacs.org/][spacemacs]] and then to
[[https://github.com/doomemacs/doomemacs][doom-emacs]] for the past decade, I've come to the conclusion that I
want to get back to vanilla emacs with my own custom setup. Following
are a few reasons why I'm doing this:

1) spacemacs and doom-emacs have a lot of baggage with opinionated editor configurations.

2) reproducibility is a pain -- I've grown accustomed to [[https://nixos.org/][Nix]] and
   reproducibility. I was using [[https://github.com/nix-ix-doom-emacs][nix-doom-emacs]] project for a while,
   but it stopped working.

3) Depedency management in emacs has gotten so much better -- Thank
   you =use-package=.

4) I like to tinker in my configuration -- exploring packages and
   trying new things out is my jam. I don't like including layers that
   are highly opinionated.

** Fancy Splash Window

I want a fancy splash window. This code assumes there is an
environment variable =SPLASH_IMG_PATH= that contains the path for the
splash image.

#+BEGIN_SRC emacs-lisp :tangle yes
  (setq initial-scratch-message nil)
  (setq inhibit-startup-screen t)

  (defun ar/show-welcome-buffer ()
    "Show *Welcome* buffer."
    (with-current-buffer (get-buffer-create "*Welcome*")
      (setq truncate-lines t)
      (let* ((buffer-read-only)
	     ;; I'm relying on the nix setup to include this SPLASH_IMG_PATH
	     (image-path (getenv "SPLASH_IMG_PATH"))
	     (image (create-image image-path))
	     (size (image-size image))
	     (height (cdr size))
	     (width (car size))
	     (top-margin (floor (/ (- (window-height) height) 2)))
	     (left-margin (floor (/ (- (window-width) width) 2)))
	     (prompt-title "Welcome to Emacs!"))
	(erase-buffer)
	(setq mode-line-format nil)
	(goto-char (point-min))
	(insert (make-string top-margin ?\n ))
	(insert (make-string left-margin ?\ ))
	(insert-image image)
	(insert "\n\n\n")
	(insert (make-string (floor (/ (- (window-width) (string-width prompt-title)) 2)) ?\ ))
	(insert prompt-title))
      (setq cursor-type nil)
      (read-only-mode +1)
      (switch-to-buffer (current-buffer))
      ;; (local-set-key (kbd "q") 'kill-this-buffer)
      (evil-local-set-key 'normal (kbd "q") #'kill-this-buffer)
      (evil-local-set-key 'insert (kbd "q") #'kill-this-buffer)))

  (when (< (length command-line-args) 2)
    (add-hook 'emacs-startup-hook (lambda ()
				    (when (display-graphic-p)
				      (ar/show-welcome-buffer)))))
#+END_SRC

*** After Save

#+begin_src emacs-lisp :tangle yes
    (defvar rapture/last-after-save-hook nil)
    (defvar rapture/hook-overrides-hash-table (make-hash-table))

  ;;;###autoload
    (defun rapture/-get-hook-funcs (hook)
      (delq nil (mapcar
		 (lambda (e) (if (symbolp e) e))
		 hook)))

  ;;;###autoload
    (defun rapture/-get-hook-funcs-names (hook)
      (mapcar 'symbol-name
	      (rapture/-get-hook-funcs
	       (if (symbolp hook)
		   (append (symbol-value hook)
			   (default-value hook))
		 hook))))

  ;;;###autoload
    (defun rapture/-get-all-hooks ()
      (let (hlist (list))
	(mapatoms (lambda (a)
		    (if (and (not (null (string-match ".*-hook"
						      (symbol-name a))))
			     (not (functionp a)))
			(add-to-list 'hlist a))))
	hlist))

  ;;;###autoload
    (defun rapture/remove-from-hook (hook fname &optional local)
      (interactive
       (let ((hook (intern (ido-completing-read
			    "Which hook? "
			    (mapcar #'symbol-name (rapture/-get-all-hooks))))))
	 (list hook
	       (ido-completing-read "Which? " (rapture/-get-hook-funcs-names hook)))))
      (remove-hook hook
		   (if (stringp fname)
		       (intern fname)
		     fname)
		   local))

  ;;;###autoload
    (defun rapture/remove-after-save-hook (fname &optional local)
      (interactive (list (ido-completing-read
			  "aWhich function: "
			  (rapture/-get-hook-funcs-names 'after-save-hook))))
      (rapture/remove-from-hook 'after-save-hook fname local))

  ;;;###autoload
    (defun rapture/add-after-save-hook (fname &optional local)
      (interactive "aWhich function: ")
      ;; write the function to the buffer-file-name for easy
      ;; removal later on
      (setf (gethash (buffer-file-name (current-buffer))
		     rapture/hook-overrides-hash-table)
	    fname)
      (setq rapture/last-after-save-hook fname)
      (add-hook 'after-save-hook fname t local)
      (message (format "%s will execute after a save" fname)))

      ;;;###autoload
    (defun rapture/add-after-save-hook-kbd (key-seq &optional local)
      (interactive
       (list (read-key-sequence "Press key: ")))
      (let ((sym (key-binding key-seq)))
	(cond
	 ((null sym)
	  (user-error "No command is bound to %s" (key-description key-seq)))
	 ((commandp sym)
	  (rapture/add-after-save-hook sym local)))))

  ;;;###autoload
    (defun rapture/check-after-save-hook (fname)
      (-contains? (rapture/-get-hook-funcs-names 'after-save-hook) (symbol-name fname)))

  ;;;###autoload
    (defun rapture/toggle-last-after-save-hook (&optional local)
      (interactive)
      (when rapture/last-after-save-hook
	(if (rapture/check-after-save-hook rapture/last-after-save-hook)
	    (progn
	      (rapture/flash-mode-line rapture/after-save-hook-disabled-color 0.5)
	      (rapture/remove-after-save-hook rapture/last-after-save-hook local))
	  ;; else
	  (rapture/flash-mode-line rapture/after-save-hook-activated-color 0.5)
	  (rapture/add-after-save-hook rapture/last-after-save-hook local))))

    (defvar rapture/after-save-hook-activated-color "#6AAF6A")
    (defvar rapture/after-save-hook-disabled-color  "#D70000")

  
    (evil-define-key 'normal 'global (kbd "SPC a h s k") #'rapture/add-after-save-hook-kbd)
    (evil-define-key 'normal 'global (kbd "SPC a h s f") #'rapture/add-after-save-hook)
    (evil-define-key 'normal 'global (kbd "SPC a h s r") #'rapture/remove-after-save-hook)
#+end_src

  (use-package ligature
    :config
    ;; Enable the "www" ligature in every possible major mode
    (ligature-set-ligatures 't '("www"))
    ;; Enable traditional ligature support in eww-mode, if the
    ;; `variable-pitch' face supports it
    (ligature-set-ligatures 'eww-mode '("ff" "fi" "ffi"))
    ;; Enable all Cascadia and Fira Code ligatures in programming modes
    (ligature-set-ligatures 'prog-mode
			  '(;; == === ==== => =| =>>=>=|=>==>> ==< =/=//=// =~
			    ;; =:= =!=
			    ("=" (rx (+ (or ">" "<" "|" "/" "~" ":" "!" "="))))
			    ;; ;; ;;;
			    (";" (rx (+ ";")))
			    ;; && &&&
			    ("&" (rx (+ "&")))
			    ;; !! !!! !. !: !!. != !== !~
			    ("!" (rx (+ (or "=" "!" "\." ":" "~"))))
			    ;; ?? ??? ?:  ?=  ?.
			    ("?" (rx (or ":" "=" "\." (+ "?"))))
			    ;; %% %%%
			    ("%" (rx (+ "%")))
			    ;; |> ||> |||> ||||> |] |} || ||| |-> ||-||
			    ;; |->>-||-<<-| |- |== ||=||
			    ;; |==>>==<<==<=>==//==/=!==:===>
			    ("|" (rx (+ (or ">" "<" "|" "/" ":" "!" "}" "\]"
					    "-" "=" ))))
			    ;; \\ \\\ \/
			    ("\\" (rx (or "/" (+ "\\"))))
			    ;; ++ +++ ++++ +>
			    ("+" (rx (or ">" (+ "+"))))
			    ;; :: ::: :::: :> :< := :// ::=
			    (":" (rx (or ">" "<" "=" "//" ":=" (+ ":"))))
			    ;; // /// //// /\ /* /> /===:===!=//===>>==>==/
			    ("/" (rx (+ (or ">"  "<" "|" "/" "\\" "\*" ":" "!"

			    ;; .. ... .... .= .- .? ..= ..<
			    ("\." (rx (or "=" "-" "\?" "\.=" "\.<" (+ "\."))))
			    ;; -- --- ---- -~ -> ->> -| -|->-->>->--<<-|
			    ("-" (rx (+ (or ">" "<" "|" "~" "-"))))
			    ;; *> */ *)  ** *** ****
			    ("*" (rx (or ">" "/" ")" (+ "*"))))
			    ;; www wwww
			    ("w" (rx (+ "w")))
			    ;; <> <!-- <|> <: <~ <~> <~~ <+ <* <$ </  <+> <*>
			    ;; <$> </> <|  <||  <||| <|||| <- <-| <-<<-|-> <->>
			    ;; <<-> <= <=> <<==<<==>=|=>==/==//=!==:=>
			    ;; << <<< <<<<

			    ;; >: >- >>- >--|-> >>-|-> >= >== >>== >=|=:=>>
			    ;; >> >>> >>>>

			    ;; #: #= #! #( #? #[ #{ #_ #_( ## ### #####
			    ("#" (rx (or ":" "=" "!" "(" "\?" "\[" "{" "_(" "_"
					 (+ "#"))))
			    ;; ~~ ~~~ ~=  ~-  ~@ ~> ~~>
			    ("~" (rx (or ">" "=" "-" "@" "~>" (+ "~"))))
			    ;; __ ___ ____ _|_ __|____|_
			    ("_" (rx (+ (or "_" "|"))))
			    ;; Fira code: 0xFF 0x12
			    ("0" (rx (and "x" (+ (in "A-F" "a-f" "0-9")))))
			    ;; Fira code:
			    "Fl"  "Tl"  "fi"  "fj"  "fl"  "ft"
			    ;; The few not covered by the regexps.
			    "{|"  "[|"  "]#"  "(*"  "}#"  "$>"  "^="))

      :config

       'highlight-indent-guides-even-face

      (set-face-background
	'highlight-indent-guides-odd-face
	(rapture/get-face-attribute 'font-lock-variable-name-face :foreground))

      ;; ensure we are using the column style
      (setq highlight-indent-guides-method 'column)

    (use-package go-mode
      :mode "\\.go"
      :hook ((go-mode . lsp-deferred)
	     (go-mode . (lambda () (setq tab-width 2))))
      :init
      (use-package gptel
	:defer t
	:init
	(defun rapture/-insert-fn-godoc (contents)
	  (when contents
	    (let* ((func-start (save-excursion (progn (go-beginning-of-defun) (point))))
		   (godoc-comment (string-trim contents))
		   (insertion-point (save-excursion
				      (goto-char func-start)
				      (line-beginning-position))))
	      (save-excursion
		(goto-char insertion-point)
		(insert godoc-comment "\n")
		(goto-char insertion-point)
		(fill-paragraph))
	      (message "Godoc comment inserted and formatted successfully."))))
	(defun rapture/gen-fn-godoc ()
	  "Generate godoc for the current Go function using gptel."
	  (interactive)
	  (if (not (eq major-mode 'go-mode))
	      (message "This function only works in go-mode")
	    (save-excursion
	      (let* ((func-start (progn (go-beginning-of-defun) (point)))
		     (func-name (progn (go-goto-function-name) (thing-at-point 'symbol)))
		     (func-end (progn (go-end-of-defun) (point)))
		     (func-content (buffer-substring-no-properties func-start func-end))
		     (prompt (concat "Generate a godoc comment for the following Go function named '"
				     func-name "', following standard library godoc standards. Focus "
                                     "on documenting 'what' the code does, not 'how' nor 'why' it "
				     "does it. Only return valid go comments, without any additional "
				     "explanation or the function itself:\n\n"
				     func-content)))
		(gptel-request prompt
		  :system "You are an expert Go programmer, skilled in writing clear and concise godoc comments."
		  :stream nil
		  :callback (lambda (response _info) (rapture/-insert-fn-godoc response)))))))
	:config
	(evil-define-key 'normal go-mode-map (kbd ", g d f") #'rapture/gen-fn-godoc)))
